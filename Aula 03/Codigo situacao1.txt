
int vermelho1 = 7;
int amarelo1 = 6;
int verde1 = 5;

int vermelho2 = 13;
int amarelo2 = 12;
int verde2 = 11;


int vermelho3 = 3;
int verde3 = 4;

void setup() {

  pinMode(vermelho1, OUTPUT);
  pinMode(amarelo1, OUTPUT);
  pinMode(verde1, OUTPUT);

  pinMode(vermelho2, OUTPUT);
  pinMode(amarelo2, OUTPUT);
  pinMode(verde2, OUTPUT);

  pinMode(vermelho3, OUTPUT);
  pinMode(verde3, OUTPUT);
}

void loop() {

  digitalWrite(verde1, 1);
  digitalWrite(amarelo1, 0);
  digitalWrite(vermelho1, 0);

  digitalWrite(verde2, 0);
  digitalWrite(amarelo2, 0);
  digitalWrite(vermelho2, 1);

  digitalWrite(verde3, 0);
  digitalWrite(vermelho3, 1);

  delay(2500);


  digitalWrite(verde1, 0);
  digitalWrite(amarelo1, 1);
  digitalWrite(vermelho1, 0);

  digitalWrite(verde2, 0);
  digitalWrite(amarelo2, 0);
  digitalWrite(vermelho2, 1);

  digitalWrite(verde3, 0);
  digitalWrite(vermelho3, 1);

  delay(500);

  
  digitalWrite(verde1, 0);
  digitalWrite(amarelo1, 0);
  digitalWrite(vermelho1, 1);

  digitalWrite(verde2, 1);
  digitalWrite(amarelo2, 0);
  digitalWrite(vermelho2, 0);

  digitalWrite(verde3, 1);
  digitalWrite(vermelho3, 0);

  delay(2500);


  digitalWrite(verde1, 0);
  digitalWrite(amarelo1, 0);
  digitalWrite(vermelho1, 1);

  digitalWrite(verde2, 0);
  digitalWrite(amarelo2, 1);
  digitalWrite(vermelho2, 0);

  digitalWrite(verde3, 1);
  digitalWrite(vermelho3, 0);

  delay(500);
}
